import math

class Matrix:
    def __init__(self, filas, columnas):
        self.filas = filas
        self.columnas = columnas
        self.matriz = [['' for _ in range(columnas)] for _ in range(filas)]
        self.filas_rellenadas = 0
        self.columnas_rellenadas = 0

    def add_letter(self, letter):
        if self.filas_rellenadas >= self.filas:
            raise ValueError("La matriz ya está llena.")

        self.matriz[self.filas_rellenadas][self.columnas_rellenadas] = letter
        self.columnas_rellenadas += 1

        if self.columnas_rellenadas == self.columnas:
            self.columnas_rellenadas = 0
            self.filas_rellenadas += 1

    def get_text(self):
        out = ''
        for fila in range(len(self.matriz)):
            for columna in range(len(self.matriz)):
                out += (self.matriz[columna][fila])
            out += " "
        return out
            
    
    def __str__(self):
        out = ''
        for fila in self.matriz:
            out += str(fila) + "\n"
        return out

        if (!$this->conn) {

        return $this->conn;

        $result = mysqli_query($this->conn, $query);

        if ($result) {
            $rows = [];
            while ($row = mysqli_fetch_assoc($result)) {
                $rows[] = $row;
            }
            return $rows;
        } else {
            return false;
        }
    }
}
